// ========================================== // 補齊：防呆自動快取與還原功能 (Local Storage) // ========================================== // 1. 自動儲存到瀏覽器快取 window.saveToLocalCache = function() { try { const data = gatherFormData(); localStorage.setItem('J57_AutoSave_Cache', JSON.stringify({ timestamp: new Date().getTime(), data: data })); } catch (e) { console.log("快取寫入暫時無法運作", e); } }; // 2. 監聽表單的任何輸入與點擊，只要有動作就自動暫存 document.addEventListener('input', window.saveToLocalCache); document.addEventListener('change', window.saveToLocalCache); // 3. 實作選單中的「從瀏覽器安全快取還原」功能 window.restoreFromLocalCache = function() { const cached = localStorage.getItem('J57_AutoSave_Cache'); if(!cached) { showToast('目前沒有暫存的進度', 'error'); return; } try { const parsed = JSON.parse(cached); // 產生一個虛擬的 ID，將暫存資料餵給載入函數 const dummyId = 'temp_cache_id'; allCustomersCache = [{ id: dummyId, ...parsed.data }]; window.loadCustomerData(dummyId); // 將 currentCustomerId 設為 null，避免不小心覆寫到雲端其他人的資料 currentCustomerId = null; showToast('已成功還原先前的編輯進度！'); } catch(e) { showToast('快取讀取失敗', 'error'); } };
